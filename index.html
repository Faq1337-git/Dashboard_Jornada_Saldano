Conexión en Tiempo Real — Google Sheets + Apps Script

Soporta las 3 hojas normalizadas (Orden_Compra, Productos, Proveedor)

Ejemplo: https://script.google.com/macros/s/.../exec

Ver código listo para pegar en Extensiones → Apps Script (Code.gs) Clic para desplegar 
function doGet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const leer = (nombre) => {
    const hoja = ss.getSheetByName(nombre);
    if (!hoja) throw new Error('Falta la hoja: ' + nombre);
    const vals = hoja.getDataRange().getValues();
    const hdrs = vals[0].map(h => String(h).trim());
    const tz = ss.getSpreadsheetTimeZone() || 'America/Argentina/Jujuy';
    return vals.slice(1).filter(r => r[0] !== '').map(r => {
      const o = {};
      hdrs.forEach((h, i) => {
        const v = r[i];
        o[h] = (v instanceof Date) ? Utilities.formatDate(v, tz, 'yyyy-MM-dd') : v;
      });
      return o;
    });
  };
  const payload = {
    status: 'ok',
    updatedAt: new Date().toISOString(),
    Orden_Compra: leer('Orden_Compra'),
    Productos: leer('Productos'),
    Proveedor: leer('Proveedor')
  };
  return ContentService.createTextOutput(JSON.stringify(payload))
    .setMimeType(ContentService.MimeType.JSON);
}
